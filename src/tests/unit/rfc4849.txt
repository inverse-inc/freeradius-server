#
#  RFC 4849 NAS-Filter-Rule
#
#  Each rule is encoded in its own attribute, terminated by a 0x00
#  byte.  The decoder concatenates the attributes and splits on 0x00,
#  as RFC 4849 describes, and devices which read one rule per
#  attribute (e.g. FortiSwitch, max 80 bytes per attribute) get one
#  rule each.
#
encode NAS-Filter-Rule = "hello"
data 5c 08 68 65 6c 6c 6f 00

decode -
data NAS-Filter-Rule = "hello"

encode NAS-Filter-Rule = "hello", NAS-Filter-Rule += "bob"
data 5c 08 68 65 6c 6c 6f 00 5c 06 62 6f 62 00

decode -
data NAS-Filter-Rule = "hello", NAS-Filter-Rule = "bob"

encode NAS-Filter-Rule = "hello", NAS-Filter-Rule += "bob", NAS-Filter-Rule += "stuff"
data 5c 08 68 65 6c 6c 6f 00 5c 06 62 6f 62 00 5c 08 73 74 75 66 66 00

decode -
data NAS-Filter-Rule = "hello", NAS-Filter-Rule = "bob", NAS-Filter-Rule = "stuff"
